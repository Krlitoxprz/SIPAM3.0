\chapter{Introduction}
% ════════════════════════════════════════════════════════════

The Universidad Surcolombiana (USCO) manages academic monitoring programs and field practice
excursions for its Faculty of Engineering under \textbf{Acuerdo 012/2023} of the Superior Council.
\textbf{SIPAM-USCO} (\textit{Sistema Integrado de Prácticas y Asesorías de Monitorías}) is the
intelligent web platform that digitizes and automates these processes through four converging technologies:

\begin{center}
\begin{tabularx}{\textwidth}{|>{\bfseries\columncolor{rowGray}}p{3cm}|X|c|}
\hline
\rowcolor{tableHdr}
\textcolor{white}{\textbf{Module}} &
\textcolor{white}{\textbf{Technology}} &
\textcolor{white}{\textbf{KPI}} \\
\hline
Machine Learning & Random Forest + MLP + Decision Tree (Scikit-learn) & F1 = 0.89 \\
\hline
CNN OCR & TensorFlow/Keras Conv2D×4, 37 alphanumeric classes & Accuracy 94.2\% \\
\hline
Transformer NLP & DistilBERT-base-uncased, 6 layers, 66 M parameters & Inference 2.8 s \\
\hline
IoT GPS & ESP32 DevKit V1 + u-blox NEO-6M, HTTP REST & Every 60 s \\
\hline
Backend / Cloud & FastAPI + PostgreSQL 16 + Docker + AWS EC2 & p95 320 ms \\
\hline
\end{tabularx}
\end{center}

\chapter{Problem Statement}
% ════════════════════════════════════════════════════════════

\section{Situational Analysis}

All monitoring and practice management relies entirely on analogue methodologies: physical forms
(FO-14, FO-46), spreadsheets, and in-person evaluations. Analysis of the current process at the
Faculty of Engineering identified six critical problems:

\begin{center}
\begin{tabularx}{\textwidth}{|>{\bfseries\columncolor{rowGray}}p{0.8cm}|p{3.8cm}|X|p{2.5cm}|}
\hline
\rowcolor{tableHdr}
\textcolor{white}{\textbf{ID}} &
\textcolor{white}{\textbf{Problem}} &
\textcolor{white}{\textbf{Description}} &
\textcolor{white}{\textbf{Impact}} \\
\hline
P1 & Manual selection & Acuerdo 012/2023 formula computed in spreadsheets, error-prone and open to manipulation & Unfair selection \\
\hline
P2 & No GPS traceability & Field excursions lack real-time location tracking & Student safety risk \\
\hline
P3 & Manual document verification & No OCR: ID cards and RUT verified by hand & Multi-day delays \\
\hline
P4 & Subjective letter evaluation & Motivation letters graded asymmetrically without semantic analysis & Evaluation bias \\
\hline
P5 & No predictive analytics & No mathematical models to predict monitor performance & Unsupported decisions \\
\hline
P6 & Fragmented workflows & Forms, QR attendance, and excursions are disconnected & Information loss \\
\hline
\end{tabularx}
\end{center}

\section{Root Cause}
\textbf{Root cause}: Total dependence on manual processes and absence of integral digital systems
for monitoring and field practice management at USCO.

\textbf{Main effects}: Loss of transparency, inability to audit, safety risk during excursions,
and asymmetry in academic evaluation.

\section{Functional Requirements — Monitoring (RF-MON)}

\begin{reqenv}{RF-MON-01}{Academic Offer Parameterization}
\reqrow{Code}{RF-MON-01}
\reqrow{Type}{Functional — \textcolor{uscoRed}{\textbf{High priority}}}
\reqrow{Description}{The Program Director creates a monitoring call with: academic period, subject (FK), type (ENUM 11 values), available slots, minimum GPA, and additional requirements. The system auto-generates a unique code.}
\reqrow{Process}{1.\ Validate JWT (role: jefe\_programa or admin). \newline 2.\ Auto-generate code \texttt{CONV-YYYYMM-XXX}. \newline 3.\ Persist with state = borrador.}
\reqrow{Criteria}{Unique code without duplicates. Creation time $\leq 3$ s. Returns HTTP 201.}
\end{reqenv}

\begin{reqenv}{RF-MON-05}{Selection Algorithm — Acuerdo 012/2023 Art.~4}
\reqrow{Code}{RF-MON-05}
\reqrow{Type}{Functional — \textcolor{uscoRed}{\textbf{Critical priority}}}
\reqrow{Description}{Applies: $0.4\bar{x}+0.3\%_{cred}+0.2\,g_{subj}+0.1\,g_{int}$. Generates ranking and invokes AI prediction in parallel. Issues FO-14 act as PDF.}
\reqrow{Process}{1.\ Retrieve all applicant evaluations. \newline 2.\ Apply official formula. \newline 3.\ Invoke Flask AI microservice. \newline 4.\ Generate ranking + FO-14 act.}
\reqrow{Criteria}{Exact Art.~4 formula. Hierarchical tie-breaking. Signed PDF is immutable.}
\end{reqenv}

\begin{reqenv}{RF-IOT-01}{GPS Telemetry — ESP32 + NEO-6M}
\reqrow{Code}{RF-IOT-01 to RF-IOT-07}
\reqrow{Type}{IoT — \textcolor{uscoRed}{\textbf{High priority}}}
\reqrow{Hardware}{ESP32 DevKit V1 (dual-core 240 MHz, WiFi 802.11 b/g/n) + u-blox NEO-6M (2.5 m CEP, 1 Hz, NMEA 0183)}
\reqrow{Protocol}{HTTP/1.1 REST POST every 60 s with header \texttt{X-Api-Key: iot\_device\_key\_12345}}
\reqrow{GPS fields}{latitude, longitude, altitude, speed\_kmh, course, satellites, hdop, datetime\_gps}
\reqrow{Telemetry}{battery\_voltage (V), wifi\_rssi (dBm), uptime\_seconds}
\reqrow{Criteria}{WiFi auto-reconnect with exponential backoff. GeoJSON route export. Public \texttt{/iot/health}.}
\end{reqenv}

\begin{reqenv}{RF-OCR-01}{CNN OCR — Text Extraction}
\reqrow{Code}{RF-OCR-01 to RF-OCR-06}
\reqrow{Type}{AI / Vision — Medium priority}
\reqrow{CNN Architecture}{Input(128×128×1)→Conv2D(32)+BN+MaxPool→Conv2D(64)+BN→Conv2D(128)+BN→Conv2D(256)+GAP→Dense(512)→Dense(37,softmax)}
\reqrow{Criteria}{37 classes (A--Z, 0--9, blank). Confidence $< 75\%$ escalates to manual review. Accuracy $\geq 90\%$.}
\end{reqenv}

\begin{reqenv}{RF-NLP-01}{Transformer NLP — Letter Analysis}
\reqrow{Code}{RF-NLP-01 to RF-NLP-06}
\reqrow{Type}{AI / NLP — \textcolor{uscoRed}{\textbf{High priority}}}
\reqrow{Description}{DistilBERT-base-uncased evaluates motivation letters: quality\_score (0--2), sentiment, coherence (0--1), key\_phrases, attention\_weights for explainability.}
\reqrow{Criteria}{Max token length: 512. Batch analysis up to 50 texts. Response time $\leq 5$ s.}
\end{reqenv}

\subsection{Acceptance Criteria — Critical Stories}

\textbf{US-12 (Application)}: System automatically validates GPA $\geq 3.5$, credits $\geq 40\%$,
and absence of sanctions before allowing inscription. Returns HTTP 400 with descriptive message if
requirements not met.

\textbf{US-20 (AI Prediction)}: Probability 0--100\% with SHAP feature importance. Comparison of
3 models. Response time $\leq 2$ s.

\textbf{US-30 (IoT GPS)}: ESP32 sends position every 60 s. Fields: lat, lng, altitude, speed,
satellites, battery, WiFi signal. History queryable and exportable as GeoJSON.

\chapter{Recommendations and Future Work}

\section{Immediate Improvements}

\begin{enumerate}[leftmargin=*]
\item \textbf{CNN OCR production model}: Train with real Colombian ID card images for higher accuracy
under low-resolution and deteriorated document conditions.
\item \textbf{DistilBERT fine-tuning}: Fine-tune on USCO motivation letter corpus for domain-specific
NLP improvement in Colombian academic Spanish.
\item \textbf{React Native mobile app}: Mobile client for students with live GPS map, push notifications,
and digital consent signatures.
\item \textbf{CI/CD pipeline}: GitHub Actions for automated testing and Docker deployment on every
push to the \texttt{main} branch.
\item \textbf{HTTPS/TLS}: Configure SSL certificate with Certbot/Let's Encrypt on the Nginx server.
\end{enumerate}

\section{Future Research Lines}

\begin{enumerate}[leftmargin=*]
\item \textbf{Federated learning}: Privacy-preserving model training across USCO faculties without
sharing student personal data.
\item \textbf{BETO (Spanish BERT)}: Domain-specific model for Colombian academic text NLP in Spanish.
\item \textbf{Document fraud detection}: CNN for anomaly detection in uploaded identity documents.
\item \textbf{IoT LoRaWAN}: Replace WiFi with LoRa for GPS tracking in areas without WiFi during
remote or rural field excursions.
\item \textbf{Multi-university SaaS}: Service model compliant with Colombia's Decreto 1075/2015.
\item \textbf{LLM report generation}: GPT/LLaMA for automatic academic report drafting for monitoring
and practice outcomes.
\end{enumerate}
